```bash
git fetch gitee develop --filter=blob:none
python riveredge-backend/scripts/sync_upstream_audit.py audit --from <水位> --to gitee/develop
# NEW=没见过 / REEVAL=评过但上游又改 / DELETED=上游删了 / SKIP=未变直接跳

```
